        #region 턴 추상화와 파라미터들

        [Serializable]
        public class TurnMessage // 턴 메세지
        {
            public string Type { get; set; } = "turnMessage"; // 메세지 타입 (turnMessage, chatMessage, ...)
            public TurnMessageType TurnMessageType { get; set; } // 턴 메세지 타입 중 하나
            public float TimerMs { get; set; } // 모든 타임은 MS로 들어옴 
            public int CurrentCycle { get; set; } // 현재 사이클
            public int CurrentRound { get; set; } // 현재 라운드

            // 타입별로 달라지는 나머지 필드 (없을 수도, JSON 객체 하나일 수도 있음)
            public string Parameter { get; set; } // 턴 메세지마다 변경되는 파라미터

            /// <summary>Parameter를 원하는 파라미터 클래스 타입으로 역직렬화. 없으면 null.</summary>
            // 밑에부터 각 타입별 파라미터 클래스가 존재하여 Getparameter로 파라미터 클래스를 읽어오면
            public T GetParameter<T>() where T : class 
            {
                if (string.IsNullOrEmpty(Parameter)) return null;
                return JsonConvert.DeserializeObject<T>(Parameter);
            }
        }
